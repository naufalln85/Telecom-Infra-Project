# -----------------------------------------------------------------------------
# CORE FRAMEWORK
# -----------------------------------------------------------------------------
# FastAPI: framework web async utama
fastapi==0.111.1

# Uvicorn: ASGI server untuk menjalankan FastAPI
# [standard] = include WebSocket support + HTTP/2

# Gunicorn: process manager untuk multi-worker di production
# Pakai bersama uvicorn: gunicorn -k uvicorn.workers.UvicornWorker
gunicorn==22.0.0

# -----------------------------------------------------------------------------
# DATABASE — PostgreSQL async
# -----------------------------------------------------------------------------
# SQLAlchemy: ORM dan query builder (versi 2.x dengan async support)
SQLAlchemy[asyncio]==2.0.30

# asyncpg: driver PostgreSQL paling cepat untuk Python async
# Dipakai oleh SQLAlchemy untuk koneksi runtime (bukan Alembic)

# psycopg2-binary: driver PostgreSQL SYNC
# Dipakai oleh Alembic untuk menjalankan migrasi (Alembic belum native async)

# Alembic: migration tool untuk SQLAlchemy
alembic==1.13.1

# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# redis-py: official Redis client untuk Python
# [hiredis]: parser C yang jauh lebih cepat dari Python parser bawaan
# Kita pakai untuk: Redis Streams (event bus), cache auth, cooldown TTL
redis[hiredis]==5.0.6

# -----------------------------------------------------------------------------
# AUTHENTICATION & SECURITY

# Kita pakai untuk generate dan verify access token setelah login

# cryptography: dibutuhkan python-jose + bisa dipakai untuk enkripsi config notifikasi
